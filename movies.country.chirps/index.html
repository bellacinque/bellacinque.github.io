  group_by(Region) |>
  mutate(Agriculture = ifelse(is.na(Agriculture), mean(Agriculture,na.rm = TRUE),    Agriculture)) |>
  mutate(Industry =ifelse(is.na(Industry), mean(Industry, na.rm = TRUE), Industry)) |>
